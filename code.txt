(function(){
  const w = $("#walker"), img = w.firstElementChild;
  if (matchMedia("(prefers-reduced-motion: reduce)").matches) return;
  let x = 0, dir = Math.random() < .5 ? 1 : -1, pause = 0, last = performance.now();
  function step(t){
    const dt = Math.min((t - last) / 1000, .1); last = t;
    const max = Math.max(0, w.parentElement.clientWidth - w.offsetWidth);
    if (pause > 0){ pause -= dt; }
    else {
      x += dir * 55 * dt;
      if (x >= max){ x = max; dir = -1; }
      if (x <= 0){ x = 0; dir = 1; }
      if (Math.random() < dt * .25) pause = 1 + Math.random() * 1.5;
    }
    img.style.animationPlayState = pause > 0 ? "paused" : "running";
    w.style.transform = "translateX(" + x + "px) scaleX(" + dir + ")";
    requestAnimationFrame(step);
  }
  requestAnimationFrame(step);
